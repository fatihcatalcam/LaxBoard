> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a full-stack Sixes lacrosse play designer — draw player paths on a half-field canvas, save/load named sets, replay as synchronized animation.

**Architecture:** Modular backend (routes → services → db) with a vanilla JS ES-module frontend (field.js, recorder.js, animator.js, sets-ui.js). All business logic lives in `setsService.js` and is tested via Jest against a real in-memory SQLite DB.

**Tech Stack:** Node.js + Express, better-sqlite3, swagger-jsdoc + swagger-ui-express, Jest, Vanilla JS ES modules, HTML Canvas.

```javascript
const Database = require('better-sqlite3');
const path = require('path');

let _db;

function getDb() {
  if (!_db) {
    _db = new Database(path.join(__dirname, '..', 'laxboard.db'));
    _db.pragma('journal_mode = WAL');
    _db.pragma('foreign_keys = ON');
    _initSchema(_db);
  }
  return _db;
}

function createTestDb() {
  const db = new Database(':memory:');
  db.pragma('foreign_keys = ON');
  _initSchema(db);
  return db;
}

function _initSchema(db) {
  db.exec(`
    CREATE TABLE IF NOT EXISTS sets (
      id         INTEGER PRIMARY KEY AUTOINCREMENT,
      name       TEXT UNIQUE NOT NULL,
      created_at TEXT DEFAULT (datetime('now')),
      updated_at TEXT DEFAULT (datetime('now'))
    );
    CREATE TABLE IF NOT EXISTS player_paths (
      id            INTEGER PRIMARY KEY AUTOINCREMENT,
      set_id        INTEGER NOT NULL REFERENCES sets(id) ON DELETE CASCADE,
      player_number INTEGER NOT NULL CHECK(player_number BETWEEN 1 AND 6),
      path          TEXT NOT NULL
    );
  `);
}

module.exports = { getDb, createTestDb };
```

```javascript
class ValidationError extends Error {
  constructor(message) {
    super(message);
    this.name = 'ValidationError';
    this.status = 400;
  }
}

class SetNotFoundError extends Error {
  constructor(id) {
    super(`Set ${id} not found`);
    this.name = 'SetNotFoundError';
    this.status = 404;
  }
}

class DuplicateNameError extends Error {
  constructor(name) {
    super(`Set name '${name}' already exists`);
    this.name = 'DuplicateNameError';
    this.status = 409;
  }
}

function validateSetName(name) {
  if (!name || typeof name !== 'string') throw new ValidationError('Name is required');
  const trimmed = name.trim();
  if (trimmed.length === 0) throw new ValidationError('Name cannot be empty');
  if (trimmed.length > 50) throw new ValidationError('Name must be 50 characters or fewer');
  if (!/^[a-zA-Z0-9ğüşıöçĞÜŞİÖÇ\s\-]+$/.test(trimmed))
    throw new ValidationError('Name may only contain letters, numbers, spaces, and hyphens');
  return trimmed;
}

function validatePaths(paths) {
  if (!Array.isArray(paths) || paths.length === 0)
    throw new ValidationError('paths must be a non-empty array');
  for (const p of paths) {
    if (!Number.isInteger(p.player_number) || p.player_number < 1 || p.player_number > 6)
      throw new ValidationError('player_number must be an integer between 1 and 6');
    if (!Array.isArray(p.path) || p.path.length === 0)
      throw new ValidationError('each path must be a non-empty array');
    for (const pt of p.path) {
      if (typeof pt.x !== 'number' || typeof pt.y !== 'number' || typeof pt.t !== 'number')
        throw new ValidationError('each point must have numeric x, y, t');
    }
  }
}

module.exports = { validateSetName, validatePaths, ValidationError, SetNotFoundError, DuplicateNameError };
```

```javascript
function createSetsService(db) {
  return {
    createSet(name) {
      const trimmed = validateSetName(name);
      try {
        const result = db.prepare('INSERT INTO sets (name) VALUES (?)').run(trimmed);
        return db.prepare('SELECT * FROM sets WHERE id = ?').get(result.lastInsertRowid);
      } catch (e) {
        if (e.message.includes('UNIQUE constraint')) throw new DuplicateNameError(trimmed);
        throw e;
      }
    },

    getSet(id) {
      const set = db.prepare('SELECT * FROM sets WHERE id = ?').get(id);
      if (!set) throw new SetNotFoundError(id);
      const rows = db.prepare('SELECT player_number, path FROM player_paths WHERE set_id = ?').all(id);
      return {
        ...set,
        paths: rows.map(r => ({ player_number: r.player_number, path: JSON.parse(r.path) }))
      };
    },

    getAllSets() {
      return db.prepare('SELECT id, name, created_at FROM sets ORDER BY created_at DESC').all();
    },

    searchSets(q) {
      if (!q || String(q).trim() === '') return this.getAllSets();
      return db.prepare(
        "SELECT id, name, created_at FROM sets WHERE name LIKE ? ORDER BY created_at DESC"
      ).all(`%${String(q).trim()}%`);
    },

    updateSet(id, name) {
      const trimmed = validateSetName(name);
      if (!db.prepare('SELECT id FROM sets WHERE id = ?').get(id)) throw new SetNotFoundError(id);
      try {
        db.prepare("UPDATE sets SET name = ?, updated_at = datetime('now') WHERE id = ?").run(trimmed, id);
        return db.prepare('SELECT * FROM sets WHERE id = ?').get(id);
      } catch (e) {
        if (e.message.includes('UNIQUE constraint')) throw new DuplicateNameError(trimmed);
        throw e;
      }
    },

    deleteSet(id) {
      if (!db.prepare('SELECT id FROM sets WHERE id = ?').get(id)) throw new SetNotFoundError(id);
      db.prepare('DELETE FROM sets WHERE id = ?').run(id);
    },

    savePaths(setId, paths) {
      validatePaths(paths);
      if (!db.prepare('SELECT id FROM sets WHERE id = ?').get(setId)) throw new SetNotFoundError(setId);
      const del = db.prepare('DELETE FROM player_paths WHERE set_id = ?');
      const ins = db.prepare('INSERT INTO player_paths (set_id, player_number, path) VALUES (?, ?, ?)');
      db.transaction(() => {
        del.run(setId);
        for (const p of paths) ins.run(setId, p.player_number, JSON.stringify(p.path));
      })();
    }
  };
}
```

```javascript
export class Field {
  constructor(canvas) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d');
    this._computeScale();
    this.players = this._defaultPositions();
    this.paths = {};   // player_number(1-6) -> [{x,y,t}]
    this.showPaths = false;
  }

  _computeScale() {
    const W = this.canvas.width, H = this.canvas.height;
    // Field is 36m wide × 35m tall (half-field)
    this._scale = Math.min(W / 36, H / 35);
    this._ox = (W - 36 * this._scale) / 2;  // x offset
    this._oy = (H - 35 * this._scale) / 2;  // y offset
  }

  // Default player positions in field meters → canvas pixels
  _defaultPositions() {
    const s = this._scale, ox = this._ox, oy = this._oy;
    return [
      [18, 18], [8, 20], [28, 20],
      [8, 28],  [18, 30], [28, 28]
    ].map(([mx, my]) => ({ x: ox + mx * s, y: oy + my * s }));
  }

  resetPlayers() {
    this.players = this._defaultPositions();
  }

  setPlayerPosition(index, x, y) {
    this.players[index] = { x, y };
  }

  // paths: [{player_number, path:[{x,y,t}]}]
  setPaths(paths) {
    this.paths = {};
    for (const p of paths) {
      this.paths[p.player_number] = p.path;
    }
  }

  setShowPaths(show) {
    this.showPaths = show;
  }

  draw() {
    const { ctx, canvas } = this;
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    this._drawField();
    if (this.showPaths) this._drawPaths();
    this._drawPlayers();
  }

  _drawField() {
    const { ctx, canvas } = this;
    const W = canvas.width, H = canvas.height;
    const s = this._scale, ox = this._ox, oy = this._oy;
    const fw = 36 * s, fh = 35 * s;

    // Grass background
    ctx.fillStyle = '#2d6b26';
    ctx.fillRect(0, 0, W, H);

    ctx.lineWidth = 2;
    ctx.strokeStyle = 'white';
    ctx.setLineDash([]);

    // End line (top)
    ctx.beginPath(); ctx.moveTo(ox, oy); ctx.lineTo(ox + fw, oy); ctx.stroke();
    // Left sideline
    ctx.beginPath(); ctx.moveTo(ox, oy); ctx.lineTo(ox, oy + fh); ctx.stroke();
    // Right sideline
    ctx.beginPath(); ctx.moveTo(ox + fw, oy); ctx.lineTo(ox + fw, oy + fh); ctx.stroke();
    // Bottom (midfield) — faint dashed
    ctx.setLineDash([8, 6]);
    ctx.strokeStyle = 'rgba(255,255,255,0.25)';
    ctx.beginPath(); ctx.moveTo(ox, oy + fh); ctx.lineTo(ox + fw, oy + fh); ctx.stroke();
    ctx.setLineDash([]);

    // Goal position: 10m from end line, centered
    const goalCX = ox + 18 * s;
    const goalY  = oy + 10 * s;
    const goalHW = 1.5 * s;   // visually enlarged half-width (real is 0.915*s)
    const goalD  = 1.5 * s;   // visually enlarged depth

    // Crease: full circle, 3m radius
    ctx.strokeStyle = 'white';
    ctx.lineWidth = 1.5;
    ctx.fillStyle = 'rgba(255,255,255,0.07)';
    ctx.beginPath();
    ctx.arc(goalCX, goalY, 3 * s, 0, Math.PI * 2);
    ctx.fill();
    ctx.stroke();

    // Goal: opens downward (faces midfield) — back is toward end line
    ctx.strokeStyle = '#ffcc00';
    ctx.lineWidth = 2.5;
    ctx.fillStyle = 'rgba(255,200,0,0.12)';
    ctx.fillRect(goalCX - goalHW, goalY - goalD, goalHW * 2, goalD);
    ctx.beginPath();
    ctx.moveTo(goalCX - goalHW, goalY);
    ctx.lineTo(goalCX - goalHW, goalY - goalD);
    ctx.lineTo(goalCX + goalHW, goalY - goalD);
    ctx.lineTo(goalCX + goalHW, goalY);
    ctx.stroke();

    // X label in the space between end line and goal
    ctx.fillStyle = 'rgba(255,255,255,0.38)';
    ctx.font = `bold ${Math.max(12, Math.round(1.2 * s))}px sans-serif`;
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.fillText('X', goalCX, oy + 5 * s);
    ctx.textBaseline = 'alphabetic';
  }

  _drawPaths() {
    const { ctx } = this;
    for (const [, path] of Object.entries(this.paths)) {
      if (path.length < 2) continue;
      ctx.strokeStyle = 'rgba(255,255,255,0.65)';
      ctx.lineWidth = 1.5;
      ctx.setLineDash([5, 4]);
      ctx.beginPath();
      ctx.moveTo(path[0].x, path[0].y);
      for (let i = 1; i < path.length; i++) ctx.lineTo(path[i].x, path[i].y);
      ctx.stroke();
      ctx.setLineDash([]);
      this._arrowHead(path[path.length - 2], path[path.length - 1]);
    }
  }

  _arrowHead(from, to) {
    const { ctx } = this;
    const angle = Math.atan2(to.y - from.y, to.x - from.x);
    const len = 10;
    ctx.fillStyle = 'rgba(255,255,255,0.8)';
    ctx.beginPath();
    ctx.moveTo(to.x, to.y);
    ctx.lineTo(to.x - len * Math.cos(angle - 0.4), to.y - len * Math.sin(angle - 0.4));
    ctx.lineTo(to.x - len * Math.cos(angle + 0.4), to.y - len * Math.sin(angle + 0.4));
    ctx.closePath();
    ctx.fill();
  }

  _drawPlayers() {
    const { ctx } = this;
    const R = 16;
    this.players.forEach((p, i) => {
      ctx.beginPath();
      ctx.arc(p.x, p.y, R, 0, Math.PI * 2);
      ctx.fillStyle = 'white';
      ctx.fill();
      ctx.strokeStyle = '#333';
      ctx.lineWidth = 1.5;
      ctx.stroke();
      ctx.fillStyle = '#111';
      ctx.font = 'bold 14px sans-serif';
      ctx.textAlign = 'center';
      ctx.textBaseline = 'middle';
      ctx.fillText(String(i + 1), p.x, p.y);
    });
    ctx.textBaseline = 'alphabetic';
  }

  // Returns the index (0-5) of the player at (x,y), or -1 if none
  getPlayerAt(x, y) {
    const R = 16;
    for (let i = 0; i < this.players.length; i++) {
      const p = this.players[i];
      if ((x - p.x) ** 2 + (y - p.y) ** 2 <= R * R) return i;
    }
    return -1;
  }
}
```

```javascript
export class Recorder {
  constructor(canvas, field, onRedraw) {
    this.canvas = canvas;
    this.field = field;
    this.onRedraw = onRedraw;

    this.state = 'IDLE';        // 'IDLE' | 'RECORDING'
    this.activePlayer = -1;     // index 0-5; -1 = none selected
    this.paths = {};            // index(0-5) -> [{x,y,t}]

    this._dragging = false;
    this._dragIdx = -1;
    this._currentPath = [];

    this._down = this._down.bind(this);
    this._move = this._move.bind(this);
    this._up   = this._up.bind(this);

    canvas.addEventListener('mousedown', this._down);
    canvas.addEventListener('mousemove', this._move);
    canvas.addEventListener('mouseup',   this._up);
  }

  startRecording() {
    this.state = 'RECORDING';
    this.activePlayer = -1;
  }

  stopRecording() {
    if (this._dragging) this._finishDrag();
    this.state = 'IDLE';
    this.activePlayer = -1;
  }

  setActivePlayer(index) {   // index 0-5
    this.activePlayer = index;
  }

  // Returns [{player_number, path}] for players that have movement paths (>1 point)
  getPaths() {
    return Object.entries(this.paths)
      .filter(([, path]) => path.length > 1)
      .map(([idx, path]) => ({ player_number: Number(idx) + 1, path }));
  }

  // Returns all 6 players' data for saving (stationary = single-point path)
  getAllPlayerData() {
    const result = [];
    for (let i = 0; i < 6; i++) {
      if (this.paths[i] && this.paths[i].length > 0) {
        result.push({ player_number: i + 1, path: this.paths[i] });
      } else {
        const p = this.field.players[i];
        result.push({ player_number: i + 1, path: [{ x: p.x, y: p.y, t: Date.now() }] });
      }
    }
    return result;
  }

  clearPaths() {
    this.paths = {};
  }

  // Load paths from a saved set (only multi-point paths stored)
  loadPaths(paths) {
    this.paths = {};
    for (const p of paths) {
      if (p.path.length > 1) {
        this.paths[p.player_number - 1] = p.path;
      }
    }
  }

  clearAll() {
    this.paths = {};
    this.field.resetPlayers();
    this.onRedraw();
  }

  _down(e) {
    const { offsetX: x, offsetY: y } = e;
    const idx = this.field.getPlayerAt(x, y);
    if (idx === -1) return;

    // In RECORDING mode, only the active player can be dragged
    if (this.state === 'RECORDING' && idx !== this.activePlayer) return;

    this._dragging = true;
    this._dragIdx = idx;
    this._currentPath = [{ x, y, t: Date.now() }];
  }

  _move(e) {
    if (!this._dragging) return;
    const { offsetX: x, offsetY: y } = e;
    this._currentPath.push({ x, y, t: Date.now() });
    this.field.setPlayerPosition(this._dragIdx, x, y);
    this.onRedraw();
  }

  _up() {
    if (!this._dragging) return;
    this._finishDrag();
  }

  _finishDrag() {
    if (this.state === 'RECORDING' && this._dragIdx === this.activePlayer) {
      this.paths[this._dragIdx] = [...this._currentPath];
    }
    this._dragging = false;
    this._dragIdx = -1;
    this._currentPath = [];
  }

  destroy() {
    this.canvas.removeEventListener('mousedown', this._down);
    this.canvas.removeEventListener('mousemove', this._move);
    this.canvas.removeEventListener('mouseup',   this._up);
  }
}
```

```javascript
export class Animator {
  constructor(field, onRedraw) {
    this.field = field;
    this.onRedraw = onRedraw;
    this._rafId = null;
    this._running = false;
  }

  // paths: [{player_number, path:[{x,y,t}]}]  (only multi-point paths)
  play(paths, onComplete) {
    if (this._running) return;

    const movingPaths = paths.filter(p => p.path.length > 1);
    if (movingPaths.length === 0) return;

    this._running = true;
    const startTime = performance.now();

    const tracks = movingPaths.map(p => ({
      idx: p.player_number - 1,
      path: p.path,
      t0: p.path[0].t,
      done: false
    }));

    const tick = (now) => {
      const elapsed = now - startTime;
      let allDone = true;

      for (const tr of tracks) {
        if (tr.done) continue;

        let lastIdx = 0;
        for (let i = 1; i < tr.path.length; i++) {
          if (tr.path[i].t - tr.t0 <= elapsed) lastIdx = i;
          else break;
        }

        const pt = tr.path[lastIdx];
        this.field.setPlayerPosition(tr.idx, pt.x, pt.y);

        if (lastIdx === tr.path.length - 1) tr.done = true;
        else allDone = false;
      }

      this.onRedraw();

      if (allDone) {
        this._running = false;
        if (onComplete) onComplete();
        return;
      }

      this._rafId = requestAnimationFrame(tick);
    };

    this._rafId = requestAnimationFrame(tick);
  }

  stop() {
    if (this._rafId) { cancelAnimationFrame(this._rafId); this._rafId = null; }
    this._running = false;
  }

  isRunning() { return this._running; }
}
```

```javascript
export class SetsUI {
  constructor({ getAllPlayerData, onLoad }) {
    this.getAllPlayerData = getAllPlayerData;
    this.onLoad = onLoad;
    this._debounce = null;

    document.getElementById('btn-save').addEventListener('click', () => this._save());
    document.getElementById('search').addEventListener('input', e => {
      clearTimeout(this._debounce);
      this._debounce = setTimeout(() => this._fetchAndRender(e.target.value), 300);
    });

    this._fetchAndRender('');
  }

  async _save() {
    const nameEl  = document.getElementById('set-name');
    const errorEl = document.getElementById('save-error');
    errorEl.textContent = '';

    const name = nameEl.value.trim();
    if (!name) { errorEl.textContent = 'Set name is required.'; return; }

    const paths = this.getAllPlayerData();

    try {
      const createRes = await fetch('/api/sets', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ name })
      });
      if (!createRes.ok) {
        const err = await createRes.json();
        errorEl.textContent = err.error;
        return;
      }
      const set = await createRes.json();

      await fetch(`/api/sets/${set.id}/paths`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ paths })
      });

      nameEl.value = '';
      this._fetchAndRender('');
    } catch {
      errorEl.textContent = 'Network error.';
    }
  }

  async _fetchAndRender(query) {
    const url = query.trim()
      ? `/api/sets/search?q=${encodeURIComponent(query.trim())}`
      : '/api/sets';
    try {
      const res = await fetch(url);
      this._render(await res.json());
    } catch { /* network failure — leave list as-is */ }
  }

  _render(sets) {
    const list = document.getElementById('sets-list');
    list.innerHTML = '';

    for (const set of sets) {
      const item = document.createElement('div');
      item.className = 'set-item';

      const nameSpan = document.createElement('span');
      nameSpan.className = 'set-name';
      nameSpan.textContent = set.name;

      const loadBtn = document.createElement('button');
      loadBtn.textContent = 'Load';
      loadBtn.addEventListener('click', () => this._load(set.id));

      const editBtn = document.createElement('button');
      editBtn.textContent = 'Edit';
      editBtn.addEventListener('click', () => this._startEdit(item, nameSpan, set));

      const delBtn = document.createElement('button');
      delBtn.textContent = 'Del';
      delBtn.addEventListener('click', () => this._delete(set.id));

      item.append(nameSpan, loadBtn, editBtn, delBtn);
      list.appendChild(item);
    }
  }

  async _load(id) {
    const res = await fetch(`/api/sets/${id}`);
    if (res.ok) this.onLoad(await res.json());
  }

  _startEdit(item, nameSpan, set) {
    const input = document.createElement('input');
    input.className = 'edit-input';
    input.value = set.name;
    nameSpan.replaceWith(input);
    input.focus();

    const commit = async () => {
      const newName = input.value.trim();
      if (!newName) { input.replaceWith(nameSpan); return; }
      const res = await fetch(`/api/sets/${set.id}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ name: newName })
      });
      if (res.ok) {
        nameSpan.textContent = newName;
        set.name = newName;
      }
      input.replaceWith(nameSpan);
      this._fetchAndRender(document.getElementById('search').value);
    };

    input.addEventListener('blur', commit);
    input.addEventListener('keydown', e => {
      if (e.key === 'Enter') commit();
      if (e.key === 'Escape') input.replaceWith(nameSpan);
    });
  }

  async _delete(id) {
    await fetch(`/api/sets/${id}`, { method: 'DELETE' });
    this._fetchAndRender(document.getElementById('search').value);
  }
}
```

```javascript
import { Field }    from './field.js';
import { Recorder } from './recorder.js';
import { Animator } from './animator.js';
import { SetsUI }   from './sets-ui.js';

const canvas    = document.getElementById('field');
const field     = new Field(canvas);
const onRedraw  = () => field.draw();
const recorder  = new Recorder(canvas, field, onRedraw);
const animator  = new Animator(field, onRedraw);

// Initial render
field.draw();

// DOM refs
const btnRecord      = document.getElementById('btn-record');
const btnStop        = document.getElementById('btn-stop');
const btnPlay        = document.getElementById('btn-play');
const btnClear       = document.getElementById('btn-clear');
const playerSelector = document.getElementById('player-selector');
const playerBtns     = document.querySelectorAll('.player-btn');
const togglePaths    = document.getElementById('toggle-paths');

// ── Record ──────────────────────────────────────────────────
btnRecord.addEventListener('click', () => {
  recorder.startRecording();
  btnRecord.disabled = true;
  btnStop.disabled   = false;
  btnPlay.disabled   = true;
  playerSelector.style.display = 'flex';
  playerSelector.style.flexDirection = 'column';
});

// ── Stop ────────────────────────────────────────────────────
btnStop.addEventListener('click', () => {
  recorder.stopRecording();
  btnRecord.disabled = false;
  btnStop.disabled   = true;
  btnPlay.disabled   = recorder.getPaths().length === 0;
  playerSelector.style.display = 'none';
  playerBtns.forEach(b => b.classList.remove('active'));
});

// ── Play ────────────────────────────────────────────────────
btnPlay.addEventListener('click', () => {
  if (animator.isRunning()) return;
  const paths = recorder.getPaths();
  // Reset players to path start positions before animating
  for (const p of paths) {
    if (p.path.length > 0)
      field.setPlayerPosition(p.player_number - 1, p.path[0].x, p.path[0].y);
  }
  field.draw();
  btnPlay.disabled = true;
  animator.play(paths, () => { btnPlay.disabled = false; });
});

// ── Clear ───────────────────────────────────────────────────
btnClear.addEventListener('click', () => {
  animator.stop();
  recorder.stopRecording();
  recorder.clearAll();
  field.setPaths([]);
  field.draw();
  btnRecord.disabled = false;
  btnStop.disabled   = true;
  btnPlay.disabled   = true;
  playerSelector.style.display = 'none';
  playerBtns.forEach(b => b.classList.remove('active'));
});

// ── Player selector ─────────────────────────────────────────
playerBtns.forEach(btn => {
  btn.addEventListener('click', () => {
    const idx = parseInt(btn.dataset.player, 10) - 1;
    recorder.setActivePlayer(idx);
    playerBtns.forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
  });
});

// ── Show Paths toggle ────────────────────────────────────────
togglePaths.addEventListener('change', () => {
  field.setShowPaths(togglePaths.checked);
  field.setPaths(recorder.getPaths());
  field.draw();
});

// ── Sets UI ─────────────────────────────────────────────────
new SetsUI({
  getAllPlayerData: () => recorder.getAllPlayerData(),
  onLoad: (set) => {
    animator.stop();
    recorder.stopRecording();

    // Restore player positions from first point of each saved path
    for (const p of set.paths) {
      if (p.path.length > 0)
        field.setPlayerPosition(p.player_number - 1, p.path[0].x, p.path[0].y);
    }

    // Load movement paths into recorder (only paths with >1 point)
    recorder.loadPaths(set.paths);

    // Sync show-paths toggle
    if (togglePaths.checked) field.setPaths(recorder.getPaths());

    field.draw();

    btnRecord.disabled = false;
    btnStop.disabled   = true;
    btnPlay.disabled   = recorder.getPaths().length === 0;
    playerSelector.style.display = 'none';
    playerBtns.forEach(b => b.classList.remove('active'));
  }
});
```

Sixes lacrosse play designer — draw player movement paths on a half-field canvas, save them as named sets, and replay as synchronized animation.

## API

Base URL: `http://localhost:3000/api`  
Swagger UI: `http://localhost:3000/api-docs`

| Method | Endpoint | Description |
|--------|----------|-------------|
| GET | `/sets` | List all sets |
| GET | `/sets/search?q=name` | Search sets by name |
| GET | `/sets/:id` | Get set with player paths |
| POST | `/sets` | Create set `{name}` |
| PUT | `/sets/:id` | Rename set `{name}` |
| DELETE | `/sets/:id` | Delete set |
| POST | `/sets/:id/paths` | Save player paths |

## Usage

1. **Record** — click Record, select a player (1–6), drag their path on the field. Repeat for other players. Click Stop.
2. **Play** — click Play to replay all recorded paths simultaneously.
3. **Show Paths** — toggle the checkbox to see path lines on the field.
4. **Save** — type a name and click Save Set.
5. **Load** — click Load next to a saved set to restore it.
6. **Edit / Delete** — use Edit to rename a set inline, Del to remove it.
